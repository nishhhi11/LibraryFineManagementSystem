cat << 'EOF' > fix_overlay.py
with open('src/GUI.java', 'r') as f:
    text = f.read()

bad_overlay = """        // 1. Overlay Panel
        JPanel overlay = new JPanel(new GridBagLayout());
        overlay.setBackground(new Color(0, 0, 0, 160)); // Dark semi-transparent
        overlay.setOpaque(true);
        overlay.setVisible(false);"""

good_overlay = """        // 1. Overlay Panel
        JPanel overlay = new JPanel(new GridBagLayout()) {
            @Override
            protected void paintComponent(Graphics g) {
                Graphics2D g2 = (Graphics2D) g.create();
                g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
                g2.setColor(new Color(0, 0, 0, 160));
                g2.fillRoundRect(0, 0, getWidth(), getHeight(), 24, 24); // match GlassPanel radius
                g2.dispose();
                super.paintComponent(g);
            }
        };
        overlay.setOpaque(false);
        overlay.setVisible(false);"""

text = text.replace(bad_overlay, good_overlay)

with open('src/GUI.java', 'w') as f:
    f.write(text)
EOF
python3 fix_overlay.py
